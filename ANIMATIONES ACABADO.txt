local function setAnimationId(animationObject, id)
	if animationObject and animationObject:IsA("Animation") then
		animationObject.AnimationId = id
	end
end

local function getAnimationId(animationObject)
	if animationObject and animationObject:IsA("Animation") then
		return animationObject.AnimationId
	end

	return nil
end

local function saveOriginalAnimations(character)
	if not character then
		return
	end

	local Animate = character:FindFirstChild("Animate")

	if not Animate then
		Animate = character:WaitForChild("Animate", 5)
	end

	if not Animate then
		return
	end

	OriginalAnimations[character] = {}

	local saved = OriginalAnimations[character]

	local Idle = Animate:FindFirstChild("idle")

	if Idle then
		saved.Idle1 = getAnimationId(Idle:FindFirstChild("Animation1"))
		saved.Idle2 = getAnimationId(Idle:FindFirstChild("Animation2"))
	end

	local Run = Animate:FindFirstChild("run")

	if Run then
		saved.Run = getAnimationId(Run:FindFirstChild("RunAnim"))
	end

	local Walk = Animate:FindFirstChild("walk")

	if Walk then
		saved.Walk = getAnimationId(Walk:FindFirstChild("WalkAnim"))
	end

	local Jump = Animate:FindFirstChild("jump")

	if Jump then
		saved.Jump = getAnimationId(Jump:FindFirstChild("JumpAnim"))
	end
end

local function restartAnimate(Animate)
	if not Animate then
		return
	end

	local oldEnabled = Animate.Enabled

	Animate.Enabled = false
	RunService.Heartbeat:Wait()
	Animate.Enabled = oldEnabled
end

local function restoreOriginalAnimations(character)
	if not character then
		return
	end

	local saved = OriginalAnimations[character]

	if not saved then
		return
	end

	local Animate = character:FindFirstChild("Animate")

	if not Animate then
		return
	end

	local Idle = Animate:FindFirstChild("idle")

	if Idle then
		setAnimationId(Idle:FindFirstChild("Animation1"), saved.Idle1)
		setAnimationId(Idle:FindFirstChild("Animation2"), saved.Idle2)
	end

	local Run = Animate:FindFirstChild("run")

	if Run then
		setAnimationId(Run:FindFirstChild("RunAnim"), saved.Run)
	end

	local Walk = Animate:FindFirstChild("walk")

	if Walk then
		setAnimationId(Walk:FindFirstChild("WalkAnim"), saved.Walk)
	end

	local Jump = Animate:FindFirstChild("jump")

	if Jump then
		setAnimationId(Jump:FindFirstChild("JumpAnim"), saved.Jump)
	end

	restartAnimate(Animate)
end

local function applyPackage(character)
	if not character or not AnimationsEnabled then
		return
	end

	local package = PACKAGES[SelectedPackage]

	if not package then
		return
	end

	local Animate = character:FindFirstChild("Animate")

	if not Animate then
		Animate = character:WaitForChild("Animate", 5)
	end

	if not Animate then
		return
	end

	if not OriginalAnimations[character] then
		saveOriginalAnimations(character)
	end

	local Idle = Animate:FindFirstChild("idle")

	if Idle then
		setAnimationId(Idle:FindFirstChild("Animation1"), package.Idle)
		setAnimationId(Idle:FindFirstChild("Animation2"), package.Idle)
	end

	local Run = Animate:FindFirstChild("run")

	if Run then
		setAnimationId(Run:FindFirstChild("RunAnim"), package.Run)
	end

	local Walk = Animate:FindFirstChild("walk")

	if Walk then
		setAnimationId(Walk:FindFirstChild("WalkAnim"), package.Run)
	end

	local Jump = Animate:FindFirstChild("jump")

	if Jump then
		setAnimationId(Jump:FindFirstChild("JumpAnim"), package.Jump)
	end

	restartAnimate(Animate)

	Status.Text = "PAQUETE: " .. SelectedPackage
end

local function updateAnimationToggle()
	if AnimationsEnabled then
		AnimationToggle.Text = "ANIMATIONS: ON"
		AnimationToggle.BackgroundColor3 = Color3.fromRGB(35, 120, 65)

		RunState.Text = "✓ RUN"
		IdleState.Text = "✓ IDLE"
		JumpState.Text = "✓ JUMP"
	else
		AnimationToggle.Text = "ANIMATIONS: OFF"
		AnimationToggle.BackgroundColor3 = Color3.fromRGB(120, 40, 40)

		RunState.Text = "✕ RUN"
		IdleState.Text = "✕ IDLE"
		JumpState.Text = "✕ JUMP"
	end
end

AnimationToggle.MouseButton1Click:Connect(function()
	AnimationsEnabled = not AnimationsEnabled

	updateAnimationToggle()

	if Player.Character then
		if AnimationsEnabled then
			applyPackage(Player.Character)
		else
			restoreOriginalAnimations(Player.Character)
		end
	end
end)

local function selectPackage(packageName)
	if not PACKAGES[packageName] then
		return
	end

	SelectedPackage = packageName

	Status.Text = "PAQUETE: " .. packageName

	updateAnimationToggle()

	if AnimationsEnabled and Player.Character then
		task.spawn(function()
			applyPackage(Player.Character)
		end)
	end
end

local MenuVisible = true

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then
		return
	end

	if input.KeyCode == Enum.KeyCode.LeftControl
		or input.KeyCode == Enum.KeyCode.RightControl then

		MenuVisible = not MenuVisible
		Main.Visible = MenuVisible
	end
end)

--==================================================
-- ARRASTRAR TODO EL MENÚ
--==================================================

local dragging = false
local dragStart
local startPosition

Main.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPosition = Main.Position
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging then
		return
	end

	if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then

		local delta = input.Position - dragStart

		Main.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = false
	end
end)
